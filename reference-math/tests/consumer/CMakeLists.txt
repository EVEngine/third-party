cmake_minimum_required(VERSION 3.21)
project(EveReferenceMathInstalledConsumer LANGUAGES C CXX)
find_package(EveReferenceMath 0.1 CONFIG REQUIRED)
add_executable(reference_math_installed_contracts ../contracts.cpp)
target_compile_features(reference_math_installed_contracts PRIVATE cxx_std_20)
add_executable(reference_math_c_consumer smoke.c)
target_compile_features(reference_math_c_consumer PRIVATE c_std_11)
# The public header is C-compatible; the archive also contains canonical C++ objects.
set_target_properties(reference_math_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
foreach(target reference_math_installed_contracts reference_math_c_consumer)
    target_link_libraries(${target} PRIVATE EveReferenceMath::eve_reference_math)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME reference-math.installed-contracts COMMAND reference_math_installed_contracts)
add_test(NAME reference-math.c-consumer COMMAND reference_math_c_consumer)
