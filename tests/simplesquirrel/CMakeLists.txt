cmake_minimum_required(VERSION 3.21)
project(NativeSimpleSquirrel LANGUAGES CXX)
include(CTest)
get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(GLOB _sq "${_root}/squirrel/squirrel/*.cpp" "${_root}/squirrel/sqstdlib/*.cpp")
file(GLOB _ssq "${_root}/simplesquirrel/source/*.cpp")
add_library(native_ssq STATIC ${_sq} ${_ssq})
target_compile_features(native_ssq PUBLIC cxx_std_20)
target_include_directories(native_ssq PUBLIC "${_root}/squirrel/include" "${_root}/simplesquirrel/include")
if(MSVC)
    target_compile_options(native_ssq PRIVATE /utf-8 /EHsc)
    target_compile_definitions(native_ssq PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    target_compile_definitions(native_ssq PUBLIC _SQ64)
endif()
add_executable(native_bindings main.cpp)
target_link_libraries(native_bindings PRIVATE native_ssq)
add_test(NAME native_bindings COMMAND native_bindings)
