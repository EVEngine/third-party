cmake_minimum_required(VERSION 3.21)
project(EVEngineTLSPackageCheck LANGUAGES C CXX)
include(CTest)
option(EVENGINE_TLS_USE_INSTALLED "Test the completed dependency install" OFF)
get_filename_component(_aggregate "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${_aggregate}/cmake/release_defaults.cmake")
if(EVENGINE_TLS_USE_INSTALLED)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    find_package(Poco REQUIRED COMPONENTS NetSSL)
else()
    include("${_aggregate}/cmake/OpenSSL.cmake")
    eve_add_openssl()
    list(PREPEND CMAKE_MODULE_PATH "${_aggregate}/cmake/bundled-openssl")
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(POCO_STATIC ON CACHE BOOL "" FORCE)
    set(POCO_DISABLE_INTERNAL_OPENSSL ON CACHE BOOL "" FORCE)
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    set(FORCE_OPENSSL ON CACHE BOOL "" FORCE)
    foreach(component NETSSL CRYPTO UTIL NET XML JSON)
        set(ENABLE_${component} ON CACHE BOOL "" FORCE)
    endforeach()
    foreach(component TESTS SAMPLES DATA DATA_SQLITE DATA_MYSQL DATA_POSTGRESQL
            DATA_ODBC MONGODB REDIS JWT ZIP PAGECOMPILER PAGECOMPILER_FILE2PAGE
            ACTIVERECORD ACTIVERECORD_COMPILER)
        set(ENABLE_${component} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${_aggregate}/poco" poco)
endif()
add_executable(tls_smoke main.cpp)
target_compile_features(tls_smoke PRIVATE cxx_std_17)
target_link_libraries(tls_smoke PRIVATE Poco::NetSSL)
if(MSVC)
    target_compile_options(tls_smoke PRIVATE /utf-8)
endif()
add_test(NAME tls.package.https COMMAND tls_smoke)
set_tests_properties(tls.package.https PROPERTIES TIMEOUT 30)
