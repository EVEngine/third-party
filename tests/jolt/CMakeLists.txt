cmake_minimum_required(VERSION 3.21)
project(EVEngineJoltPackageCheck LANGUAGES CXX)
include(CTest)
option(EVENGINE_JOLT_USE_INSTALLED "Test an installed Jolt::Jolt package" OFF)
get_filename_component(_aggregate "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(EVENGINE_JOLT_USE_INSTALLED)
    find_package(Jolt CONFIG REQUIRED)
else()
    # Sentinels verify upstream option names do not leak into the parent.
    set(BUILD_SHARED_LIBS ON)
    set(DOUBLE_PRECISION OFF)
    set(OVERRIDE_CXX_FLAGS ON)
    include("${_aggregate}/cmake/Jolt.cmake")
    eve_add_jolt()
    if(NOT BUILD_SHARED_LIBS OR DOUBLE_PRECISION OR NOT OVERRIDE_CXX_FLAGS)
        message(FATAL_ERROR "Jolt configuration leaked into the parent scope")
    endif()
    if(NOT EVENGINE_BUILD_JOLT)
        if(TARGET Jolt)
            message(FATAL_ERROR "Disabled Jolt unexpectedly created a target")
        endif()
        return()
    endif()
    get_target_property(_type Jolt TYPE)
    if(NOT _type STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "Packaged Jolt must be static")
    endif()
endif()
get_target_property(_definitions Jolt::Jolt INTERFACE_COMPILE_DEFINITIONS)
if(NOT "JPH_DOUBLE_PRECISION" IN_LIST _definitions)
    message(FATAL_ERROR "Jolt package did not export its double-precision ABI")
endif()
add_executable(jolt_smoke "${_aggregate}/jolt/HelloWorld/HelloWorld.cpp")
target_compile_features(jolt_smoke PRIVATE cxx_std_17)
target_link_libraries(jolt_smoke PRIVATE Jolt::Jolt)
add_test(NAME jolt.package.falling_body COMMAND jolt_smoke)
set_tests_properties(jolt.package.falling_body PROPERTIES
    PASS_REGULAR_EXPRESSION "A body went to sleep" TIMEOUT 30)
